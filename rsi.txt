//@version=6
// Version: 1.0.1
indicator(title="Relative Strength Index v1.0.1", shorttitle="RSI v1.0.1", format=format.price, precision=2)

rsiLengthInput = input.int(14, minval=1, title="RSI Length", group="RSI Settings")
invertRSI = input.bool(false, title="Invert RSI (100 - RSI)", group="RSI Settings")
rsiSourceInput = input.source(close, "Source", group="RSI Settings")
calculateDivergence = input.bool(false, title="Calculate Divergence", group="RSI Settings", display = display.data_window, tooltip = "Calculating divergences is needed in order for divergence alerts to fire.")

change = ta.change(rsiSourceInput)
up = ta.rma(math.max(change, 0), rsiLengthInput)
down = ta.rma(-math.min(change, 0), rsiLengthInput)
rsi_raw = down == 0 ? 100 : up == 0 ? 0 : 100 - (100 / (1 + up / down))
rsi = invertRSI ? (100 - rsi_raw) : rsi_raw

rsiPlot = plot(rsi, "RSI", color=#7E57C2)
rsiUpperBand  = hline(60, "RSI Upper Band",  color=#787B86)
rsiUpperBand2 = hline(80, "RSI Upper Band 2", color=color.new(#787B86, 40), linestyle=hline.style_dashed)
midline       = hline(50, "RSI Middle Band", color=color.new(#787B86, 50))
rsiLowerBand  = hline(40, "RSI Lower Band",  color=#787B86)
rsiLowerBand2 = hline(20, "RSI Lower Band 2", color=color.new(#787B86, 40), linestyle=hline.style_dashed)
fill(rsiUpperBand, rsiLowerBand, color=color.rgb(126, 87, 194, 90), title="RSI Background Fill")
midLinePlot = plot(50, color = na, editable = false, display = display.none)
fill(rsiPlot, midLinePlot, 100, 80, top_color = color.new(color.green, 0), bottom_color = color.new(color.green, 100),  title = "Overbought Gradient Fill")
fill(rsiPlot, midLinePlot, 20,  0,  top_color = color.new(color.red, 100), bottom_color = color.new(color.red, 0),      title = "Oversold Gradient Fill")

// Smoothing MA inputs
GRP = "Smoothing"
TT_BB = "Only applies when 'SMA + Bollinger Bands' is selected. Determines the distance between the SMA and the bands."
maTypeInput = input.string("SMA", "Type", options = ["None", "SMA", "SMA + Bollinger Bands", "EMA", "SMMA (RMA)", "WMA", "VWMA"], group = GRP, display = display.data_window)
maLengthInput = input.int(14, "Length", group = GRP, display = display.data_window)
bbMultInput = input.float(2.0, "BB StdDev", minval = 0.001, maxval = 50, step = 0.5, tooltip = TT_BB, group = GRP, display = display.data_window)
var enableMA = maTypeInput != "None"
var isBB = maTypeInput == "SMA + Bollinger Bands"

// Smoothing MA Calculation
ma(source, length, MAtype) =>
    switch MAtype
        "SMA"                   => ta.sma(source, length)
        "SMA + Bollinger Bands" => ta.sma(source, length)
        "EMA"                   => ta.ema(source, length)
        "SMMA (RMA)"            => ta.rma(source, length)
        "WMA"                   => ta.wma(source, length)
        "VWMA"                  => ta.vwma(source, length)

// Smoothing MA plots
smoothingMA = enableMA ? ma(rsi, maLengthInput, maTypeInput) : na
smoothingStDev = isBB ? ta.stdev(rsi, maLengthInput) * bbMultInput : na
plot(smoothingMA, "RSI-based MA", color=color.yellow, display = enableMA ? display.all : display.none, editable = enableMA)
bbUpperBand = plot(smoothingMA + smoothingStDev, title = "Upper Bollinger Band", color=color.green, display = isBB ? display.all : display.none, editable = isBB)
bbLowerBand = plot(smoothingMA - smoothingStDev, title = "Lower Bollinger Band", color=color.green, display = isBB ? display.all : display.none, editable = isBB)
fill(bbUpperBand, bbLowerBand, color= isBB ? color.new(color.green, 90) : na, title="Bollinger Bands Background Fill", display = isBB ? display.all : display.none, editable = isBB)

// ─── EMA 9 & WMA 45 trên RSI ───────────────────────────────────────────────
GRP2 = "EMA / WMA on RSI"
showEMA9  = input.bool(true,  title="Show EMA 9",  group=GRP2)
showWMA45 = input.bool(true,  title="Show WMA 45", group=GRP2)
ema9Length  = input.int(9,  minval=1, title="EMA Length",  group=GRP2)
wma45Length = input.int(45, minval=1, title="WMA Length",  group=GRP2)

ema9  = ta.ema(rsi, ema9Length)
wma45 = ta.wma(rsi, wma45Length)

plot(showEMA9  ? ema9  : na, title="EMA 9 on RSI",  color=color.new(color.aqua, 0), linewidth=1)
plot(showWMA45 ? wma45 : na, title="WMA 45 on RSI", color=color.new(color.orange, 0), linewidth=1)

// ─── Crossover / Crossunder alerts ─────────────────────────────────────────
emaCrossOver  = ta.crossover(ema9, wma45)
emaCrossUnder = ta.crossunder(ema9, wma45)

alertcondition(emaCrossOver,  title="EMA 9 cắt lên WMA 45",   message="RSI: EMA 9 vừa cắt lên trên WMA 45 – tín hiệu tăng.")
alertcondition(emaCrossUnder, title="EMA 9 cắt xuống WMA 45", message="RSI: EMA 9 vừa cắt xuống dưới WMA 45 – tín hiệu giảm.")

// ─── Multi-Timeframe Settings ───────────────────────────────────────────────
GRP3 = "Multi-Timeframe Settings"
showTable = input.bool(true,  title="Hiển thị bảng xu hướng MTF", group=GRP3)
tf4x      = input.timeframe("60",  title="Khung x4",  group=GRP3)
tf16x     = input.timeframe("240", title="Khung x16", group=GRP3)

// Hàm tính RSI thủ công để dùng trong request.security
f_rsi(src, len) =>
    chg = ta.change(src)
    u   = ta.rma(math.max(chg, 0), len)
    d   = ta.rma(-math.min(chg, 0), len)
    d == 0 ? 100 : u == 0 ? 0 : 100 - (100 / (1 + u / d))

// Lấy EMA và WMA từ khung x4
[ema9_4x, wma45_4x] = request.security(syminfo.tickerid, tf4x, [
     ta.ema(f_rsi(close, rsiLengthInput), ema9Length),
     ta.wma(f_rsi(close, rsiLengthInput), wma45Length)
     ], lookahead=barmerge.lookahead_off)

// Lấy EMA và WMA từ khung x16
[ema9_16x, wma45_16x] = request.security(syminfo.tickerid, tf16x, [
     ta.ema(f_rsi(close, rsiLengthInput), ema9Length),
     ta.wma(f_rsi(close, rsiLengthInput), wma45Length)
     ], lookahead=barmerge.lookahead_off)

// Xác định xu hướng
trend4x  = ema9_4x  > wma45_4x  ? "▲ Tăng" : ema9_4x  < wma45_4x  ? "▼ Giảm" : "─ Ngang"
trend16x = ema9_16x > wma45_16x ? "▲ Tăng" : ema9_16x < wma45_16x ? "▼ Giảm" : "─ Ngang"

color4x  = ema9_4x  > wma45_4x  ? color.new(color.lime, 10) : ema9_4x  < wma45_4x  ? color.new(color.red, 10) : color.new(color.gray, 10)
color16x = ema9_16x > wma45_16x ? color.new(color.lime, 10) : ema9_16x < wma45_16x ? color.new(color.red, 10) : color.new(color.gray, 10)

// ─── Hiển thị bảng góc phải dưới ──────────────────────────────────────────
var table infoTable = table.new(position.bottom_right, 2, 3, border_width=1, border_color=color.new(color.gray, 50), frame_color=color.new(color.gray, 30), frame_width=1)

if barstate.islast
    if showTable
        table.cell(infoTable, 0, 0, "Khung TF", text_color=color.white, bgcolor=color.new(color.navy, 20), text_size=size.small)
        table.cell(infoTable, 1, 0, "EMA/WMA",  text_color=color.white, bgcolor=color.new(color.navy, 20), text_size=size.small)

        table.cell(infoTable, 0, 1, "x4 (" + tf4x  + ")", text_color=color.white, bgcolor=color.new(color.gray, 70), text_size=size.small)
        table.cell(infoTable, 1, 1, trend4x,  text_color=color.white, bgcolor=color4x,  text_size=size.small)

        table.cell(infoTable, 0, 2, "x16 (" + tf16x + ")", text_color=color.white, bgcolor=color.new(color.gray, 70), text_size=size.small)
        table.cell(infoTable, 1, 2, trend16x, text_color=color.white, bgcolor=color16x, text_size=size.small)
    else
        table.clear(infoTable, 0, 0, 1, 2)

// ─── Divergence ────────────────────────────────────────────────────────────
lookbackRight = 5
lookbackLeft  = 5
rangeUpper    = 60
rangeLower    = 5
bearColor     = color.red
bullColor     = color.green
textColor     = color.white
noneColor     = color.new(color.white, 100)

_inRange(bool cond) =>
    bars = ta.barssince(cond)
    rangeLower <= bars and bars <= rangeUpper

plFound = false
phFound = false
bullCond = false
bearCond = false

rsiLBR = rsi[lookbackRight]

if calculateDivergence
    plFound := not na(ta.pivotlow(rsi, lookbackLeft, lookbackRight))
    rsiHL = rsiLBR > ta.valuewhen(plFound, rsiLBR, 1) and _inRange(plFound[1])
    lowLBR = low[lookbackRight]
    priceLL = lowLBR < ta.valuewhen(plFound, lowLBR, 1)
    bullCond := priceLL and rsiHL and plFound

    phFound := not na(ta.pivothigh(rsi, lookbackLeft, lookbackRight))
    rsiLH = rsiLBR < ta.valuewhen(phFound, rsiLBR, 1) and _inRange(phFound[1])
    highLBR = high[lookbackRight]
    priceHH = highLBR > ta.valuewhen(phFound, highLBR, 1)
    bearCond := priceHH and rsiLH and phFound

plot(plFound ? rsiLBR : na, offset=-lookbackRight, title="Regular Bullish",       linewidth=2, color=(bullCond ? bullColor : noneColor), display=display.pane, editable=calculateDivergence)
plotshape(bullCond ? rsiLBR : na, offset=-lookbackRight, title="Regular Bullish Label", text=" Bull ", style=shape.labelup,   location=location.absolute, color=bullColor, textcolor=textColor, display=display.pane, editable=calculateDivergence)
plot(phFound ? rsiLBR : na, offset=-lookbackRight, title="Regular Bearish",       linewidth=2, color=(bearCond ? bearColor : noneColor), display=display.pane, editable=calculateDivergence)
plotshape(bearCond ? rsiLBR : na, offset=-lookbackRight, title="Regular Bearish Label", text=" Bear ", style=shape.labeldown, location=location.absolute, color=bearColor, textcolor=textColor, display=display.pane, editable=calculateDivergence)

alertcondition(bullCond, title='Regular Bullish Divergence',  message="Found a new Regular Bullish Divergence, `Pivot Lookback Right` number of bars to the left of the current bar.")
alertcondition(bearCond, title='Regular Bearish Divergence',  message='Found a new Regular Bearish Divergence, `Pivot Lookback Right` number of bars to the left of the current bar.')

// ─── Alert: 2 nến liên tiếp dưới 20 / trên 80 ────────────────────────────
twoCandlesBelow20 = rsi[1] < 20 and rsi < 20
twoCandlesAbove80 = rsi[1] > 80 and rsi > 80

alertcondition(twoCandlesBelow20, title="2 nến liên tiếp dưới RSI 20", message="RSI: 2 nến liên tiếp dưới mức 20 – vùng quá bán cực mạnh.")
alertcondition(twoCandlesAbove80, title="2 nến liên tiếp trên RSI 80", message="RSI: 2 nến liên tiếp trên mức 80 – vùng quá mua cực mạnh.")
