strategy("SMC - breakout - retest v1.0.5", shorttitle="SMC Strategy v1.0.5", overlay=true, initial_capital=10000, default_qty_type=strategy.percent_of_equity, default_qty_value=10, commission_type=strategy.commission.percent, commission_value=0.04, slippage=1, pyramiding=0)

// ==== CẤU HÌNH THỊ TRƯỜNG & TÍN HIỆU ====
swingSize = input.int(25, "Market Structure Time-Horizon", tooltip='Defines the number of candles used to determine market structure. Larger values smooth out market structure changes but may delay signals.', group="Market Structure")
bosConfType = input.string('Candle Close', 'BOS Confirmation Type', ['Candle Close', 'Wicks'], tooltip='Specifies whether a candle close or a wick surpassing the previous swing point is used to confirm a Break of Structure (BOS).', group="Market Structure")
choch = input.bool(true, 'Show CHoCH', tooltip='Enables or disables the display of Change of Character (CHoCH) labels, highlighting the first structural break against the trend.', group="Market Structure")

// ==== QUẢN LÝ LỆNH & RỦI RO (RISK MANAGEMENT) ====
riskPerTradePct = input.float(1.0, "Risk mỗi lệnh (%)", minval=0.01, step=0.1, tooltip="Tỷ lệ % rủi ro trên tổng tài khoản (Equity) cho mỗi lệnh (ví dụ: 1% của $10k là $100, khi tài khoản tăng $11k là $110)", group="Risk & Strategy")
rrRatio = input.float(2.0, "Risk / Reward Ratio (RR)", minval=0.1, step=0.1, tooltip="Tùy chọn tỷ lệ Risk:Reward cho chốt lời (TP = Entry +/- Dist * RR)", group="Risk & Strategy")
cancelOnSL = input.bool(true, "Cancel Retest If Price Invalidates SL", tooltip="Cancels the pending retest if price moves beyond Stop Loss before retesting.", group="Risk & Strategy")
cancelOnTP = input.bool(false, "Cancel Retest If Price Reaches TP First", tooltip="Cancels the pending retest if price already reaches TP before retesting.", group="Risk & Strategy")

// ==== BỘ LỌC THỜI GIAN BACKTEST ====
useDateFilter = input.bool(false, "Filter Date Range", tooltip="Giới hạn khoảng thời gian thực hiện backtest", group="Backtest Time Range")
startDate = input.time(timestamp("2023-01-01 00:00"), "Start Date", group="Backtest Time Range")
endDate = input.time(timestamp("2026-12-31 23:59"), "End Date", group="Backtest Time Range")
inDateRange = not useDateFilter or (time >= startDate and time <= endDate)

// ==== HIỂN THỊ & BẢNG THỐNG KÊ ====
showSL = input.bool(true, 'Show Stop Loss Lines', tooltip='Enables or disables the display of Stop Loss lines on chart.', group="Visualization")
showRetestLine = input.bool(true, "Show Pending Retest Line", tooltip="Displays a line at the breakout level waiting for price to retest.", group="Visualization")
showTable = input.bool(true, "Show Performance Table", tooltip="Enables or disables the display of performance and drawdown statistics table.", group="Visualization")
color BULL = input.color(#00ffbb, "Bullish Color", group="Visualization", tooltip='Sets the color used to represent bullish signals and breakout lines.')
color BEAR = input.color(#ff1100, "Bearish Color", group="Visualization", tooltip='Sets the color used to represent bearish signals and breakout lines.')
color SL_COLOR = input.color(#ff6600, "Stop Loss Color", group="Visualization", tooltip='Sets the color used for Stop Loss lines.')

MS(int swingSize, int src_BI, string bosConfType, bool enms, float src_close, float src_high, float src_low) =>
    pivHi = ta.pivothigh(src_high, swingSize, swingSize)
    pivLo = ta.pivotlow(src_low, swingSize, swingSize)
    var float prevHigh = na
    var float prevLow = na
    var int prevHighIndex = na
    var int prevLowIndex = na
    var bool highActive = false
    var bool lowActive = false
    bool hh = false
    bool lh = false
    bool hl = false
    bool ll = false
    var int prevSwing = 0
    if not na(pivHi) and enms
        if pivHi >= prevHigh
            hh := true
            prevSwing := 2
        else
            lh := true
            prevSwing := 1
        prevHigh := pivHi
        highActive := true
        prevHighIndex := src_BI - swingSize
    if not na(pivLo) and enms
        if pivLo >= prevLow
            hl := true
            prevSwing := -1
        else
            ll := true
            prevSwing := -2
        prevLow := pivLo
        lowActive := true
        prevLowIndex := src_BI - swingSize
    bool highBroken = false
    bool lowBroken = false
    float highSrc = bosConfType == 'Candle Close' ? src_close : src_high
    float lowSrc = bosConfType == 'Candle Close' ? src_close : src_low
    if highSrc > prevHigh and highActive and enms
        highBroken := true
        highActive := false
    if lowSrc < prevLow and lowActive and enms
        lowBroken := true
        lowActive := false
    [hh,ll,lh,hl,highBroken,lowBroken,prevHighIndex,prevLowIndex,pivHi,pivLo,prevHigh,prevLow]

[hh, ll, lh, hl, highBroken, lowBroken, prevHighIndex, prevLowIndex, pivHi, pivLo, prevHigh, prevLow] = MS(swingSize, bar_index, bosConfType, true, close, high, low)

var int prevBreakoutDir = 0
length = 0

if highBroken
    line.new(prevHighIndex, prevHigh, bar_index, prevHigh, color=BULL, width=1, force_overlay=true)
    label.new(math.floor(bar_index - (bar_index - prevHighIndex) / 2), prevHigh, prevBreakoutDir == -1 and choch ? 'CHoCH' : 'BOS', color=color.rgb(0, 0, 0, 100), textcolor=BULL, size=size.tiny, force_overlay=true)
    alert("Bullish Breakout Detected", alert.freq_once_per_bar)
    prevBreakoutDir := 1
    length := bar_index - prevHighIndex
if lowBroken
    line.new(prevLowIndex, prevLow, bar_index, prevLow, color=BEAR, width=1, force_overlay=true)
    label.new(math.floor(bar_index - (bar_index - prevLowIndex) / 2), prevLow, prevBreakoutDir == 1 and choch ? 'CHoCH' : 'BOS', color=color.rgb(0, 0, 0, 100), textcolor=BEAR, style=label.style_label_up, size=size.tiny, force_overlay=true)
    alert("Bearish Breakout Detected", alert.freq_once_per_bar)
    prevBreakoutDir := -1
    length := bar_index - prevLowIndex

x = math.max(1, nz(length))
v = ta.highest(x) - ta.lowest(x)
dist = v / 3
trigger = 0

// Tùy chọn RR (Risk:Reward)
RR = rrRatio

// Hàm tính toán khối lượng vào lệnh (position sizing) dựa trên % rủi ro tài khoản hiện tại
calcQty(float riskDist) =>
    riskCash = strategy.equity * (riskPerTradePct / 100.0)
    pv = nz(syminfo.pointvalue, 1.0)
    qty = (riskDist > 0 and pv > 0) ? (riskCash / (riskDist * pv)) : na
    qty

var active = 0
var active1 = 0
var active2 = 0
var TP = 0.0
var TP1 = 0.0
var TP2 = 0.0
var SL = 0.0
var TPs = array.new_line()
var TPs1 = array.new_line()
var TPs2 = array.new_line()
var SLs = array.new_line()  // array cho Stop Loss lines

var total = 0
var tp1s = 0
var tp2s = 0
var tp3s = 0

// Thống kê hiệu suất, Drawdown & Chuỗi lệnh
var float netProfitR = 0.0
var float peakProfitR = 0.0
var float maxDrawdownR = 0.0
var float equity = 100.0        // Giả định vốn xuất phát 100%
var float peakEquity = 100.0
var float maxDrawdownPct = 0.0
var int currentConsecLosses = 0
var int maxConsecLosses = 0
var int totalWins = 0
var int totalLosses = 0

// ==== TRẠNG THÁI CHỜ RETEST (PENDING RETEST) ====
var int pendingSignal = 0       // 1: Chờ Bullish Retest, -1: Chờ Bearish Retest, 0: Không có
var float pendingLevel = na     // Mức giá cần retest (prevHigh hoặc prevLow)
var float pendingDist = na      // Khoảng cách dist tại thời điểm breakout
var int pendingBarIndex = na    // Bar index phát sinh tín hiệu breakout
var line pendingLine = na       // Đường hiển thị mức retest đang chờ

// Khi có tín hiệu Breakout mới: Cập nhật trạng thái chờ Retest (Bỏ qua tín hiệu cũ nếu chưa retest)
if highBroken
    if not na(pendingLine)
        pendingLine.delete()
    pendingSignal := 1
    pendingLevel := prevHigh
    pendingDist := dist
    pendingBarIndex := bar_index
    if showRetestLine
        pendingLine := line.new(bar_index, pendingLevel, bar_index, pendingLevel, color=color.new(BULL, 30), width=1, style=line.style_dotted, force_overlay=true)

if lowBroken
    if not na(pendingLine)
        pendingLine.delete()
    pendingSignal := -1
    pendingLevel := prevLow
    pendingDist := dist
    pendingBarIndex := bar_index
    if showRetestLine
        pendingLine := line.new(bar_index, pendingLevel, bar_index, pendingLevel, color=color.new(BEAR, 30), width=1, style=line.style_dotted, force_overlay=true)

// Kéo dài đường chờ retest theo các nến tiếp theo
if pendingSignal != 0 and not na(pendingLine)
    pendingLine.set_x2(bar_index)

// ==== KIỂM TRA ĐIỀU KIỆN RETEST VÀO LỆNH (ENTRY) ====
bool buyEntry = false
bool sellEntry = false

if pendingSignal == 1 and bar_index > pendingBarIndex
    if cancelOnSL and low < (pendingLevel - pendingDist) and high < pendingLevel
        // Giá rơi thủng qua Stop Loss trước khi kịp retest -> Bỏ qua tín hiệu
        pendingSignal := 0
        if not na(pendingLine)
            pendingLine.delete()
            pendingLine := na
    else if cancelOnTP and high >= (pendingLevel + pendingDist * RR)
        // Giá đã đạt TP trước khi retest -> Bỏ qua tín hiệu
        pendingSignal := 0
        if not na(pendingLine)
            pendingLine.delete()
            pendingLine := na
    else if low <= pendingLevel
        // Giá retest lại mức đỉnh đã phá vỡ thành công -> Kích hoạt Entry
        buyEntry := true
        pendingSignal := 0
        if not na(pendingLine)
            pendingLine.delete()
            pendingLine := na

if pendingSignal == -1 and bar_index > pendingBarIndex
    if cancelOnSL and high > (pendingLevel + pendingDist) and low > pendingLevel
        // Giá tăng vượt qua Stop Loss trước khi kịp retest -> Bỏ qua tín hiệu
        pendingSignal := 0
        if not na(pendingLine)
            pendingLine.delete()
            pendingLine := na
    else if cancelOnTP and low <= (pendingLevel - pendingDist * RR)
        // Giá đã đạt TP trước khi retest -> Bỏ qua tín hiệu
        pendingSignal := 0
        if not na(pendingLine)
            pendingLine.delete()
            pendingLine := na
    else if high >= pendingLevel
        // Giá retest lại mức đáy đã phá vỡ thành công -> Kích hoạt Entry
        sellEntry := true
        pendingSignal := 0
        if not na(pendingLine)
            pendingLine.delete()
            pendingLine := na

// ==== VÀO LỆNH KHI RETEST THÀNH CÔNG (LONG) ====
if buyEntry
    active := 1
    active1 := 1
    active2 := 1
    trigger := 1
    total += 1

    // SL = 1R, TP = RR (tính từ mức retest)
    SL  := pendingLevel - pendingDist
    TP  := pendingLevel + pendingDist * RR
    TP1 := pendingLevel + pendingDist * RR * 2 / 3
    TP2 := pendingLevel + pendingDist * RR * 1 / 3

    if inDateRange
        tradeQty = calcQty(pendingDist)
        strategy.entry("Long", strategy.long, qty=tradeQty)
        strategy.exit("Exit Long", from_entry="Long", stop=SL, limit=TP)

    TPs.unshift(line.new(bar_index, TP, bar_index, TP, color=color.new(BULL, 70), width=1, force_overlay=true, style=line.style_dashed))
    TPs1.unshift(line.new(bar_index, TP1, bar_index, TP1, color=color.new(BULL, 50), width=1, force_overlay=true, style=line.style_dashed))
    TPs2.unshift(line.new(bar_index, TP2, bar_index, TP2, color=color.new(BULL, 30), width=1, force_overlay=true, style=line.style_dashed))

    if showSL
        SLs.unshift(line.new(bar_index, SL, bar_index, SL, color=color.new(SL_COLOR, 30), width=2, force_overlay=true, style=line.style_solid))

    linefill.new(TPs.first(), TPs1.first(), color.new(BULL, 90))
    alert("Bullish Retest Entry Executed", alert.freq_once_per_bar)

// ==== VÀO LỆNH KHI RETEST THÀNH CÔNG (SHORT) ====
if sellEntry
    active := -1
    active1 := -1
    active2 := -1
    trigger := 1
    total += 1

    // SL = 1R, TP = RR (tính từ mức retest)
    SL  := pendingLevel + pendingDist
    TP  := pendingLevel - pendingDist * RR
    TP1 := pendingLevel - pendingDist * RR * 2 / 3
    TP2 := pendingLevel - pendingDist * RR * 1 / 3

    if inDateRange
        tradeQty = calcQty(pendingDist)
        strategy.entry("Short", strategy.short, qty=tradeQty)
        strategy.exit("Exit Short", from_entry="Short", stop=SL, limit=TP)

    TPs.unshift(line.new(bar_index, TP, bar_index, TP, color=color.new(BEAR, 70), width=1, force_overlay=true, style=line.style_dashed))
    TPs1.unshift(line.new(bar_index, TP1, bar_index, TP1, color=color.new(BEAR, 50), width=1, force_overlay=true, style=line.style_dashed))
    TPs2.unshift(line.new(bar_index, TP2, bar_index, TP2, color=color.new(BEAR, 30), width=1, force_overlay=true, style=line.style_dashed))

    if showSL
        SLs.unshift(line.new(bar_index, SL, bar_index, SL, color=color.new(SL_COLOR, 30), width=2, force_overlay=true, style=line.style_solid))

    linefill.new(TPs.first(), TPs1.first(), color.new(BEAR, 90))
    alert("Bearish Retest Entry Executed", alert.freq_once_per_bar)

// Duy trì lệnh Stop Loss và Take Profit liên tục khi vị thế đang mở
if strategy.position_size > 0
    strategy.exit("Exit Long", from_entry="Long", stop=SL, limit=TP)
if strategy.position_size < 0
    strategy.exit("Exit Short", from_entry="Short", stop=SL, limit=TP)

// ==== TP HIT LOGIC (CHO BẢNG THỐNG KÊ TÍCH HỢP) ====
if active == 1 and high > TP
    tp1s += 1
if active1 == 1 and high > TP1
    tp2s += 1
    active1 := 0
if active2 == 1 and high > TP2
    tp3s += 1
    active2 := 0

if active == -1 and low < TP
    tp1s += 1
if active1 == -1 and low < TP1
    tp2s += 1
    active1 := 0
if active2 == -1 and low < TP2
    tp3s += 1
    active2 := 0

// Đóng lệnh khi TP hoặc SL chạm & Cập nhật Drawdown / Lợi nhuận
bool isEntryBar = trigger != 0 and trigger[1] == 0
bool isLongClose = active == 1 and not isEntryBar and (high > TP or low < SL)
bool isShortClose = active == -1 and not isEntryBar and (low < TP or high > SL)

if isLongClose or isShortClose
    bool isWin = false
    if isLongClose
        if high > TP and low < SL
            // Cả TP và SL cùng quét trong 1 nến: kiểm tra khoảng cách từ giá mở cửa
            isWin := (open - SL) > (TP - open)
        else
            isWin := high > TP
    else if isShortClose
        if low < TP and high > SL
            isWin := (SL - open) > (open - TP)
        else
            isWin := low < TP

    float tradeProfitR = isWin ? RR : -1.0
    netProfitR += tradeProfitR
    peakProfitR := math.max(peakProfitR, netProfitR)
    float currentDDR = peakProfitR - netProfitR
    maxDrawdownR := math.max(maxDrawdownR, currentDDR)

    float tradeReturnRate = tradeProfitR * (riskPerTradePct / 100.0)
    equity := equity * (1.0 + tradeReturnRate)
    peakEquity := math.max(peakEquity, equity)
    float currentDDPct = peakEquity > 0 ? ((peakEquity - equity) / peakEquity) * 100.0 : 0.0
    maxDrawdownPct := math.max(maxDrawdownPct, currentDDPct)

    if isWin
        totalWins += 1
        currentConsecLosses := 0
    else
        totalLosses += 1
        currentConsecLosses += 1
        maxConsecLosses := math.max(maxConsecLosses, currentConsecLosses)

    active := 0

// Tự động đóng toàn bộ lệnh chiến lược nếu quá khoảng thời gian backtest
if useDateFilter and time > endDate
    strategy.close_all(comment="End of Backtest Period")

// Quản lý TP lines
if TPs.size() > 0
    if active == 0 or (active != 0 and active[1] != 0 and trigger == 1)
        TPs.remove(math.min(1, TPs.size() - 1))
    else
        TPs.first().set_x2(bar_index)

if TPs1.size() > 0
    if active == 0 or (active != 0 and active[1] != 0 and trigger == 1)
        TPs1.remove(math.min(1, TPs1.size() - 1))
    else
        TPs1.first().set_x2(bar_index)

if TPs2.size() > 0
    if active == 0 or (active != 0 and active[1] != 0 and trigger == 1)
        TPs2.remove(math.min(1, TPs2.size() - 1))
    else
        TPs2.first().set_x2(bar_index)

// Quản lý hiển thị đường Stop Loss
if SLs.size() > 0 and showSL
    if active == 0 or (active != 0 and active[1] != 0 and trigger == 1)
        SLs.remove(math.min(1, SLs.size() - 1))
    else
        SLs.first().set_x2(bar_index)

barcolor(active[0] == 0 ? color.new(color.gray, 40) : active[0] == 1 ? color.new(BULL[1], 40) : color.new(BEAR[1], 40))

// Tín hiệu Breakout ban đầu (Dấu tròn nhỏ)
plotshape(highBroken, "Bullish Breakout", shape.circle, location.belowbar, color.new(BULL, 30), size=size.tiny)
plotshape(lowBroken, "Bearish Breakout", shape.circle, location.abovebar, color.new(BEAR, 30), size=size.tiny)

// Tín hiệu vào lệnh khi Retest thành công
plotshape(buyEntry, "Bullish Retest Entry", shape.labelup, location.belowbar, BULL, text="BUY", textcolor=color.black, size=size.small)
plotshape(sellEntry, "Bearish Retest Entry", shape.labeldown, location.abovebar, BEAR, text="SELL", textcolor=color.white, size=size.small)

if barstate.islast
    var table data_table = table.new(
         position=position.top_right,
         columns=2,
         rows=9,
         bgcolor=chart.bg_color,
         border_width=1,
         border_color=chart.fg_color,
         frame_color=chart.fg_color,
         frame_width=1
     )

    if showTable
        t1_rate = total > 0 ? (tp1s / total) * 100.0 : na
        t2_rate = total > 0 ? (tp2s / total) * 100.0 : na
        t3_rate = total > 0 ? (tp3s / total) * 100.0 : na

        profitColor = netProfitR > 0 ? color.lime : netProfitR < 0 ? color.red : chart.fg_color
        ddColor = maxDrawdownR > 0 ? color.red : chart.fg_color

        table.cell(
             data_table,
             column   = 0,
             row      = 0,
             text     = "Smart Money Breakout Stats",
             text_size= size.tiny,
             text_color = chart.fg_color,
             text_halign= text.align_center
         )
        table.cell(
             data_table,
             column   = 1,
             row      = 0,
             text     = "Value",
             text_size= size.tiny,
             text_color = chart.fg_color,
             text_halign= text.align_center
         )

        table.cell(
             data_table,
             column   = 0,
             row      = 1,
             text     = "Total Trades (Retest)",
             text_size= size.tiny,
             text_color = chart.fg_color
         )
        table.cell(
             data_table,
             column   = 1,
             row      = 1,
             text     = str.tostring(total),
             text_size= size.tiny,
             text_color = chart.fg_color,
             text_halign= text.align_center
         )

        table.cell(
             data_table,
             column   = 0,
             row      = 2,
             text     = "TP1 Win Rate",
             text_size= size.tiny,
             text_color = chart.fg_color
         )
        table.cell(
             data_table,
             column   = 1,
             row      = 2,
             text     = total > 0 ? str.format("{0,number,#.##}%", t3_rate) : "No Trades",
             text_size= size.tiny,
             text_color = chart.fg_color,
             text_halign= text.align_center
         )

        table.cell(
             data_table,
             column   = 0,
             row      = 3,
             text     = "TP2 Win Rate",
             text_size= size.tiny,
             text_color = chart.fg_color
         )
        table.cell(
             data_table,
             column   = 1,
             row      = 3,
             text     = total > 0 ? str.format("{0,number,#.##}%", t2_rate) : "No Trades",
             text_size= size.tiny,
             text_color = chart.fg_color,
             text_halign= text.align_center
         )

        table.cell(
             data_table,
             column   = 0,
             row      = 4,
             text     = "TP3 Win Rate",
             text_size= size.tiny,
             text_color = chart.fg_color
         )
        table.cell(
             data_table,
             column   = 1,
             row      = 4,
             text     = total > 0 ? str.format("{0,number,#.##}%", t1_rate) : "No Trades",
             text_size= size.tiny,
             text_color = chart.fg_color,
             text_halign= text.align_center
         )

        table.cell(
             data_table,
             column   = 0,
             row      = 5,
             text     = "Net Profit (R)",
             text_size= size.tiny,
             text_color = chart.fg_color
         )
        table.cell(
             data_table,
             column   = 1,
             row      = 5,
             text     = total > 0 ? (netProfitR > 0 ? "+" : "") + str.format("{0,number,#.##} R", netProfitR) : "No Trades",
             text_size= size.tiny,
             text_color = profitColor,
             text_halign= text.align_center
         )

        table.cell(
             data_table,
             column   = 0,
             row      = 6,
             text     = "Max Drawdown (R)",
             text_size= size.tiny,
             text_color = chart.fg_color
         )
        table.cell(
             data_table,
             column   = 1,
             row      = 6,
             text     = total > 0 ? (maxDrawdownR > 0 ? "-" : "") + str.format("{0,number,#.##} R", maxDrawdownR) : "No Trades",
             text_size= size.tiny,
             text_color = ddColor,
             text_halign= text.align_center
         )

        table.cell(
             data_table,
             column   = 0,
             row      = 7,
             text     = "Max Drawdown (%)",
             text_size= size.tiny,
             text_color = chart.fg_color
         )
        table.cell(
             data_table,
             column   = 1,
             row      = 7,
             text     = total > 0 ? (maxDrawdownPct > 0 ? "-" : "") + str.format("{0,number,#.##}%", maxDrawdownPct) : "No Trades",
             text_size= size.tiny,
             text_color = ddColor,
             text_halign= text.align_center
         )

        table.cell(
             data_table,
             column   = 0,
             row      = 8,
             text     = "Max Consec Losses",
             text_size= size.tiny,
             text_color = chart.fg_color
         )
        table.cell(
             data_table,
             column   = 1,
             row      = 8,
             text     = total > 0 ? str.tostring(maxConsecLosses) : "No Trades",
             text_size= size.tiny,
             text_color = chart.fg_color,
             text_halign= text.align_center
         )
    else
        table.clear(data_table, 0, 0, 1, 8)

//////Alerts
